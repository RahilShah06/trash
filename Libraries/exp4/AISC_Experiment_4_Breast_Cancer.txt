# BREAST CANCER CLASSIFICATION USING PERCEPTRON

import numpy as np
import matplotlib.pyplot as plt
from sklearn.datasets import load_breast_cancer
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import Perceptron
from sklearn.metrics import (
    accuracy_score,
    confusion_matrix,
    ConfusionMatrixDisplay,
    classification_report
)

print("=" * 65)
print("       BREAST CANCER DATASET - PERCEPTRON")
print("=" * 65)

# Load the built-in Breast Cancer Wisconsin Diagnostic dataset
data = load_breast_cancer()
X = data.data
y = data.target

print("\nDATASET INFORMATION")
print("Total Samples     :", X.shape[0])
print("Total Features    :", X.shape[1])
print("Number of Classes :", len(np.unique(y)))
print("0 = Malignant")
print("1 = Benign")

# Convert labels to bipolar values: Malignant = +1, Benign = -1
y_bipolar = np.where(y == 0, 1, -1)

# Split into 80% training and 20% testing data
X_train, X_test, y_train, y_test = train_test_split(
    X,
    y_bipolar,
    test_size=0.20,
    random_state=42,
    stratify=y_bipolar
)

print("\nTRAINING AND TESTING DATA")
print("Training Samples:", X_train.shape[0])
print("Testing Samples :", X_test.shape[0])

# Standardize using training data only
scaler = StandardScaler()
X_train = scaler.fit_transform(X_train)
X_test = scaler.transform(X_test)

print("\nFEATURE PREPROCESSING")
print("Standardization: Applied")
print("Mean: Approximately 0")
print("Standard Deviation: Approximately 1")

# Train the Perceptron
model = Perceptron(
    max_iter=1000,
    eta0=0.01,
    tol=1e-3,
    random_state=42
)

print("\nTraining Perceptron...")
model.fit(X_train, y_train)
print("Training completed successfully.")

# Predict test samples
y_pred = model.predict(X_test)
accuracy = accuracy_score(y_test, y_pred)

print("\nMODEL PERFORMANCE")
print("Accuracy:", round(accuracy, 4))
print("Accuracy Percentage:", round(accuracy * 100, 2), "%")

print("\nCLASSIFICATION REPORT")
print(classification_report(
    y_test,
    y_pred,
    labels=[1, -1],
    target_names=["Malignant", "Benign"],
    digits=4
))

cm = confusion_matrix(y_test, y_pred, labels=[1, -1])
print("\nCONFUSION MATRIX")
print(cm)

disp = ConfusionMatrixDisplay(
    confusion_matrix=cm,
    display_labels=["Malignant", "Benign"]
)
fig, ax = plt.subplots(figsize=(7, 6))
disp.plot(ax=ax, values_format="d")
plt.title("Breast Cancer - Perceptron Confusion Matrix")
plt.xlabel("Predicted Label")
plt.ylabel("Actual Label")
plt.tight_layout()
plt.show()

print("\nSAMPLE PREDICTIONS")
for i in range(min(10, len(X_test))):
    actual = "Malignant" if y_test[i] == 1 else "Benign"
    predicted = "Malignant" if y_pred[i] == 1 else "Benign"
    print(f"Sample {i + 1:02d} | Actual: {actual:<10} | Predicted: {predicted}")

print("\nEXPERIMENT COMPLETED")
print("Final Test Accuracy:", round(accuracy * 100, 2), "%")
