# OR GATE USING ADALINE LEARNING RULE
# This section follows the additional OR-gate calculation in the uploaded Excel workbook.

# Bipolar input patterns and target values
X = [
    [1, 1],
    [1, -1],
    [-1, 1],
    [-1, -1]
]
t = [1, 1, 1, -1]

# Initial values from the workbook
w1 = 0.1
w2 = 0.1
b = 0.1
alpha = 0.1

# The workbook displays two training epochs
epochs = 2

print("\nOR GATE USING ADALINE LEARNING RULE")
print("Epoch X1 X2 t Yin Error DeltaW1 DeltaW2 DeltaB NewW1 NewW2 NewB SquaredError")

for epoch in range(1, epochs + 1):
    for i in range(len(X)):
        x1, x2 = X[i]
        target = t[i]

        # ADALINE uses the continuous net input as its output
        yin = b + w1 * x1 + w2 * x2
        error = target - yin

        # Least Mean Squares (LMS) updates
        dw1 = alpha * error * x1
        dw2 = alpha * error * x2
        db = alpha * error

        w1 = w1 + dw1
        w2 = w2 + dw2
        b = b + db

        squared_error = error ** 2

        print(epoch, x1, x2, target,
              round(yin, 6), round(error, 6),
              round(dw1, 6), round(dw2, 6), round(db, 6),
              round(w1, 6), round(w2, 6), round(b, 6),
              round(squared_error, 8))

print("\nWeights after", epochs, "epochs")
print("W1 =", round(w1, 9))
print("W2 =", round(w2, 9))
print("Bias =", round(b, 9))
