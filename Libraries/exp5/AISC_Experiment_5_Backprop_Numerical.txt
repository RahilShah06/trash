# EXPERIMENT 5: BACKPROPAGATION NUMERICAL EXAMPLE
# 2 INPUTS -> 2 HIDDEN NEURONS -> 1 OUTPUT NEURON

import numpy as np
import pandas as pd

# Input values and target
x1 = 0.0
x2 = 1.0
target = 1.0
alpha = 0.25

# Input -> Hidden layer weights and biases
v11 = 0.6
v21 = -0.1
v01 = 0.3

v12 = -0.3
v22 = 0.4
v02 = 0.5

# Hidden -> Output layer weights and bias
w1 = 0.4
w2 = 0.1
w0 = -0.2

# Sigmoid activation function
def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))

results = []

# Perform 10 iterations
for iteration in range(1, 11):

    # Forward propagation: hidden neuron z1
    z1_in = v01 + (x1 * v11) + (x2 * v21)
    z1 = sigmoid(z1_in)

    # Forward propagation: hidden neuron z2
    z2_in = v02 + (x1 * v12) + (x2 * v22)
    z2 = sigmoid(z2_in)

    # Forward propagation: output neuron
    y_in = w0 + (z1 * w1) + (z2 * w2)
    y = sigmoid(y_in)

    # Output layer error term
    delta_y = (target - y) * y * (1 - y)

    # Hidden layer error terms
    delta_z1 = delta_y * w1 * z1 * (1 - z1)
    delta_z2 = delta_y * w2 * z2 * (1 - z2)

    # Changes in hidden-to-output weights
    delta_w1 = alpha * delta_y * z1
    delta_w2 = alpha * delta_y * z2
    delta_w0 = alpha * delta_y

    # Changes in input-to-hidden weights
    delta_v11 = alpha * delta_z1 * x1
    delta_v21 = alpha * delta_z1 * x2
    delta_v01 = alpha * delta_z1

    delta_v12 = alpha * delta_z2 * x1
    delta_v22 = alpha * delta_z2 * x2
    delta_v02 = alpha * delta_z2

    # Save iteration details before updating weights
    results.append({
        "Iteration": iteration,
        "z1_in": z1_in,
        "z1": z1,
        "z2_in": z2_in,
        "z2": z2,
        "y_in": y_in,
        "y": y,
        "delta_y": delta_y,
        "delta_z1": delta_z1,
        "delta_z2": delta_z2,
        "Delta_w1": delta_w1,
        "Delta_w2": delta_w2,
        "Delta_w0": delta_w0,
        "Delta_v11": delta_v11,
        "Delta_v21": delta_v21,
        "Delta_v01": delta_v01,
        "Delta_v12": delta_v12,
        "Delta_v22": delta_v22,
        "Delta_v02": delta_v02
    })

    # Update hidden-to-output weights and bias
    w1 = w1 + delta_w1
    w2 = w2 + delta_w2
    w0 = w0 + delta_w0

    # Update input-to-hidden weights and biases
    v11 = v11 + delta_v11
    v21 = v21 + delta_v21
    v01 = v01 + delta_v01

    v12 = v12 + delta_v12
    v22 = v22 + delta_v22
    v02 = v02 + delta_v02

# Display iteration table
df = pd.DataFrame(results)
pd.set_option("display.max_columns", None)
pd.set_option("display.width", 220)

print("\n" + "=" * 120)
print("BACKPROPAGATION - 10 ITERATIONS")
print("=" * 120)
print(df[["Iteration", "z1_in", "z1", "z2_in", "z2", "y_in", "y", "delta_y", "delta_z1", "delta_z2"]].round(6).to_string(index=False))

print("\n" + "=" * 120)
print("WEIGHT CHANGES")
print("=" * 120)
print(df[["Iteration", "Delta_w1", "Delta_w2", "Delta_w0", "Delta_v11", "Delta_v21", "Delta_v01", "Delta_v12", "Delta_v22", "Delta_v02"]].round(6).to_string(index=False))

print("\n" + "=" * 120)
print("FINAL WEIGHTS AFTER 10 ITERATIONS")
print("=" * 120)
print(f"v11 = {v11:.6f}")
print(f"v21 = {v21:.6f}")
print(f"v01 = {v01:.6f}")
print(f"v12 = {v12:.6f}")
print(f"v22 = {v22:.6f}")
print(f"v02 = {v02:.6f}")
print(f"w1 = {w1:.6f}")
print(f"w2 = {w2:.6f}")
print(f"w0 = {w0:.6f}")
