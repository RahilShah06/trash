import numpy as np
import pandas as pd
import matplotlib.pyplot as plt


def sigmoid(x):
    x = np.clip(x, -500, 500)
    return 1 / (1 + np.exp(-x))


def train_network(X, y, feature_names, case_name):
    # Split data into 80% training and 20% testing
    rng = np.random.default_rng(42)
    indices = rng.permutation(len(X))
    split = int(0.8 * len(X))

    train_idx = indices[:split]
    test_idx = indices[split:]

    X_train = X[train_idx]
    X_test = X[test_idx]
    y_train = y[train_idx]
    y_test = y[test_idx]

    # Standardize inputs using training data only
    X_mean = X_train.mean(axis=0)
    X_std = X_train.std(axis=0)
    X_std[X_std == 0] = 1

    X_train = (X_train - X_mean) / X_std
    X_test = (X_test - X_mean) / X_std

    # Neural network: input layer -> 8 hidden neurons -> 1 output neuron
    input_neurons = X.shape[1]
    hidden_neurons = 8

    learning_rate = 0.20
    momentum = 0.50
    epochs = 2000

    weight_rng = np.random.default_rng(109)
    W1 = weight_rng.normal(0, 0.25, size=(input_neurons, hidden_neurons))
    b1 = np.zeros((1, hidden_neurons))
    W2 = weight_rng.normal(0, 0.25, size=(hidden_neurons, 1))
    b2 = np.zeros((1, 1))

    # Momentum values start at zero
    V_W1 = np.zeros_like(W1)
    V_b1 = np.zeros_like(b1)
    V_W2 = np.zeros_like(W2)
    V_b2 = np.zeros_like(b2)

    history = []

    # Backpropagation training
    for epoch in range(1, epochs + 1):
        # Forward propagation
        Z1 = X_train @ W1 + b1
        A1 = sigmoid(Z1)
        Z2 = A1 @ W2 + b2
        output = sigmoid(Z2)

        # Mean squared error
        error = y_train - output
        mse = np.mean(error ** 2)
        history.append(mse)

        # Backward propagation (sigmoid + MSE)
        delta_output = error * output * (1 - output)
        delta_hidden = (delta_output @ W2.T) * A1 * (1 - A1)

        # Average gradients across the training samples
        dW2 = A1.T @ delta_output / len(X_train)
        db2 = np.sum(delta_output, axis=0, keepdims=True) / len(X_train)
        dW1 = X_train.T @ delta_hidden / len(X_train)
        db1 = np.sum(delta_hidden, axis=0, keepdims=True) / len(X_train)

        # Momentum-based weight updates
        V_W2 = momentum * V_W2 + learning_rate * dW2
        V_b2 = momentum * V_b2 + learning_rate * db2
        V_W1 = momentum * V_W1 + learning_rate * dW1
        V_b1 = momentum * V_b1 + learning_rate * db1

        W2 = W2 + V_W2
        b2 = b2 + V_b2
        W1 = W1 + V_W1
        b1 = b1 + V_b1

    # Calculate final training MSE after the last weight update
    final_train_output = sigmoid(sigmoid(X_train @ W1 + b1) @ W2 + b2)
    final_train_mse = np.mean((y_train - final_train_output) ** 2)

    # Test the trained network
    hidden_test = sigmoid(X_test @ W1 + b1)
    probabilities = sigmoid(hidden_test @ W2 + b2)
    predicted = (probabilities >= 0.5).astype(int)

    accuracy = np.mean(predicted == y_test) * 100

    # Confusion matrix: rows = actual, columns = predicted
    cm = np.zeros((2, 2), dtype=int)
    for actual, guess in zip(y_test.flatten(), predicted.flatten()):
        cm[int(actual), int(guess)] += 1

    print("\n" + "=" * 65)
    print(case_name)
    print("=" * 65)
    print("Total samples       :", len(X))
    print("Training samples    :", len(X_train))
    print("Testing samples     :", len(X_test))
    print("Input features      :", input_neurons)
    print("Hidden neurons      :", hidden_neurons)
    print("Training epochs     :", epochs)
    print("Learning rate       :", learning_rate)
    print("Momentum            :", momentum)
    print("Initial MSE         :", round(history[0], 6))
    print("Final training MSE  :", round(final_train_mse, 6))
    print("Test accuracy       :", round(accuracy, 2), "%")

    print("\nConfusion Matrix")
    print("                 Predicted 0  Predicted 1")
    print("Actual 0            ", cm[0, 0], "          ", cm[0, 1])
    print("Actual 1            ", cm[1, 0], "          ", cm[1, 1])

    # Display a few test predictions
    original_test = X[ test_idx ]
    sample_data = pd.DataFrame(original_test[:10], columns=feature_names)
    sample_data["Actual"] = y_test[:10].flatten()
    sample_data["Predicted"] = predicted[:10].flatten()
    sample_data["Probability of class 1"] = probabilities[:10].flatten().round(4)

    print("\nSample Test Predictions (1 = positive class, 0 = negative class)")
    print(sample_data.to_string(index=False))

    # Plot training error
    plt.figure(figsize=(8, 5))
    plt.plot(range(1, epochs + 1), history)
    plt.xlabel("Epoch")
    plt.ylabel("Mean Squared Error")
    plt.title(case_name + " - Training Error")
    plt.grid(True)
    plt.tight_layout()
    plt.show()

    # Plot confusion matrix
    plt.figure(figsize=(5, 4))
    plt.imshow(cm)
    plt.title(case_name + " - Confusion Matrix")
    plt.xlabel("Predicted Class")
    plt.ylabel("Actual Class")
    plt.xticks([0, 1], ["0", "1"])
    plt.yticks([0, 1], ["0", "1"])
    for row in range(2):
        for col in range(2):
            plt.text(col, row, str(cm[row, col]), ha="center", va="center")
    plt.tight_layout()
    plt.show()

# CASE STUDY 1: STUDENT PASS / FAIL PREDICTION
# Inputs: study hours and attendance
# Output: 1 = Pass, 0 = Fail

rng = np.random.default_rng(42)
N = 80

study_hours = rng.uniform(0, 10, N)
attendance = rng.uniform(50, 100, N)

# Dummy teaching rule used to create labels
passed = ((study_hours >= 5) & (attendance >= 70)) |          ((study_hours >= 8) & (attendance >= 60))

y = passed.astype(int).reshape(-1, 1)
X = np.column_stack((study_hours, attendance))

train_network(
    X, y,
    ["Study_Hours", "Attendance"],
    "STUDENT PASS / FAIL PREDICTION"
)
