# CARTESIAN PRODUCT OF FUZZY SETS

import numpy as np

A = np.array([0.3, 0.7, 1.0])
B = np.array([0.4, 0.9])

R = np.zeros((len(A), len(B)))

for i in range(len(A)):
    for j in range(len(B)):
        R[i, j] = min(A[i], B[j])

print("Set A:", A)
print("Set B:", B)
print("\nCartesian Product / Fuzzy Relation:")
print(R)
